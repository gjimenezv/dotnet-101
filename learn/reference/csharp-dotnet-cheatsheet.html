OOP: constructor, interface, herencia

public interface IPaymentProcessor
{
    Task<bool> ProcessAsync(decimal amount);
}

public abstract class Shape
{
    public abstract double GetArea();          // sin cuerpo, obligatorio
    public void Print() => Console.WriteLine(GetArea());
}

public class Circle : Shape                     // ": Shape" = herencia
{
    public double Radius { get; set; }
    public Circle(double radius) { Radius = radius; }   // constructor
    public override double GetArea() => Math.PI * Radius * Radius;
}

Controller mínimo

[ApiController]
[Route("api/[controller]")]
public class ProductsController : ControllerBase
{
    private readonly IProductService _service;
    public ProductsController(IProductService service) => _service = service;

    [HttpGet]
    public async Task<ActionResult<List<ProductDto>>> GetAll()
        => Ok(await _service.GetAllAsync());

    [HttpGet("{id}")]
    public async Task<ActionResult<ProductDto>> GetById(int id)
    {
        var p = await _service.GetByIdAsync(id);
        return p is null ? NotFound() : Ok(p);
    }

    [HttpPost]
    public async Task<ActionResult<ProductDto>> Create(CreateProductDto input)
    {
        var created = await _service.CreateAsync(input);
        return CreatedAtAction(nameof(GetById), new { id = created.Id }, created);
    }

    [HttpDelete("{id}")]
    public async Task<IActionResult> Delete(int id)
        => await _service.DeleteAsync(id) ? NoContent() : NotFound();
}
